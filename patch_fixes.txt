rad_dedup AS (
    SELECT * EXCEPT(rn)
    FROM (
        SELECT
            b.*,
            ROW_NUMBER() OVER (
                PARTITION BY b.claimsummaryedenuid, CAST(b.selectionmonth AS STRING)
                ORDER BY b.audit_selecteddate DESC NULLS LAST
            ) AS rn

    )
    WHERE rn = 1

candidate_matches AS (

        a.client,
        a.claimsummaryuid,
        a.month_run,
        a.selection_type AS selection_type_b,
        b.*,
        CASE
            WHEN CAST(b.selectionmonth AS STRING) = a.month_run THEN 0
            WHEN v2.selected = 1 AND CAST(b.selectionmonth AS STRING) = CAST(v2.audit_selecteddate AS STRING) THEN 0
            ELSE 1
        END AS is_offset,
        ROW_NUMBER() OVER (
            PARTITION BY b.claimsummaryedenuid, CAST(b.selectionmonth AS STRING), a.selection_type
            ORDER BY
                CASE
                    WHEN CAST(b.selectionmonth AS STRING) = a.month_run THEN 0
                    WHEN v2.selected = 1 THEN 1
                    ELSE 2
                END ASC,

            CAST(b.selectionmonth AS STRING) IN (
                a.month_run,
                DATE_FORMAT(ADD_MONTHS(TO_DATE(a.month_run, 'yyyyMM'), 1), 'yyyyMM')
            )
            OR (v2.selected = 1 AND CAST(b.selectionmonth AS STRING) = CAST(v2.audit_selecteddate AS STRING))
            OR (v2.selected = 1)
        )
),

rad_matched AS (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY claimsummaryuid, month_run, selection_type_b
            ORDER BY is_offset ASC, audit_selecteddate DESC NULLS LAST
        ) AS rad_rn
    FROM candidate_matches
    WHERE (rad_claim_rn = 1 OR v2_is_selected = 1)
      AND (
          is_offset = 0
          OR v2_is_selected = 1
          OR (is_offset = 1 AND cohort_exact_matches = 0)
      )
)

All-Client Validation Query (Run After EXP_Report_mart_UHC Finishes)
Run this single query to check all clients (UHCCOSMOS, UHCCIRRUS, UHCUNET, UHCUMR, UHCCSP) at once—it verifies that UHCCOSMOS (<= 202607) matches report_mart_v2 and all other client-months (UHCCIRRUS, UHCUNET, UHCUMR, and UHCCOSMOS 202608+) match cleanly:
%sql
WITH exp_data AS (
    SELECT
        UPPER(client) AS client,
        CAST(month_run AS STRING) AS month_run,
        COUNT(DISTINCT DataSourceProcessLogId) AS exp_dspld_cnt,
        FIRST(DataSourceProcessLogId) AS exp_dlp,
        SUM(inflow) AS exp_inflow,
        SUM(inscope) AS exp_inscope,
        SUM(selected) AS exp_selected,
        SUM(approved) AS exp_approved,
        SUM(true_finding) AS exp_true_finding
    FROM mldev.uhcunet_gold.report_mart_exp
    WHERE selection_type = 'DRG' AND month_run >= '202509'
    GROUP BY UPPER(client), CAST(month_run AS STRING)
),
ref_data AS (
    -- Benchmark: report_mart_v2 for UHCCOSMOS <= 202607; report_mart for everything else
    SELECT
        UPPER(client) AS client,
        CAST(month_run AS STRING) AS month_run,
        FIRST(DataSourceProcessLogId) AS ref_dlp,
        SUM(inflow) AS ref_inflow,
        SUM(inscope) AS ref_inscope,
        SUM(selected) AS ref_selected,
        SUM(approved) AS ref_approved,
        SUM(true_finding) AS ref_true_finding
    FROM mldev.uhcunet_gold.report_mart_v2
    WHERE selection_type = 'DRG' AND UPPER(client) = 'UHCCOSMOS' AND month_run BETWEEN '202509' AND '202607'
    GROUP BY UPPER(client), CAST(month_run AS STRING)

    UNION ALL

    SELECT
        UPPER(client) AS client,
        CAST(month_run AS STRING) AS month_run,
        FIRST(DataSourceProcessLogId) AS ref_dlp,
        SUM(inflow) AS ref_inflow,
        SUM(inscope) AS ref_inscope,
        SUM(selected) AS ref_selected,
        SUM(approved) AS ref_approved,
        SUM(true_finding) AS ref_true_finding
    FROM mldev.uhcunet_gold.report_mart
    WHERE selection_type = 'DRG'
      AND month_run >= '202509'
      AND NOT (UPPER(client) = 'UHCCOSMOS' AND month_run <= '202607')
    GROUP BY UPPER(client), CAST(month_run AS STRING)
)
SELECT
    COALESCE(e.client, r.client) AS client,
    COALESCE(e.month_run, r.month_run) AS month_run,
    e.exp_dspld_cnt,
    e.exp_dlp,
    e.exp_inflow - r.ref_inflow AS diff_inflow,
    e.exp_inscope - r.ref_inscope AS diff_inscope,
    e.exp_selected - r.ref_selected AS diff_selected,
    e.exp_approved - r.ref_approved AS diff_approved,
    e.exp_true_finding,
    r.ref_true_finding
FROM exp_data e
FULL OUTER JOIN ref_data r
    ON e.client = r.client AND e.month_run = r.month_run
ORDER BY client, month_run DESC;
