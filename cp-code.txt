from concurrent.futures import ThreadPoolExecutor, as_completed

# Create lists to track query execution
success_queries = []
error_queries = []
result_list = []

# Extract the rows we need to process into a local list
rows_to_process = updated_temp_rules.collect()

def execute_rule(row):
    """Function to execute a single query and return its results."""
    query = row.RuleDef2
    try:
        # Execute the query
        result_df = spark.sql(query)
        # Collect the results immediately to force execution
        result_rows = [str(r[0]) for r in result_df.collect()]
        
        return {
            "status": "success",
            "legacy_name": row.LegacyQueryName,
            "rule_id": row.ruleid,
            "rule_type_id": row.ExpressionRuleTypeid,
            "data": result_rows
        }
    except Exception as e:
        return {
            "status": "error",
            "legacy_name": row.LegacyQueryName,
            "rule_id": row.ruleid
        }

# Execute concurrently using 8 worker threads 
# (Adjust max_workers based on your cluster's capacity)
with ThreadPoolExecutor(max_workers=8) as executor:
    # Submit all tasks to the executor
    futures = {executor.submit(execute_rule, row): row for row in rows_to_process}
    
    # Process results as they complete (order doesn't matter for the final table)
    for future in as_completed(futures):
        result = future.result()
        
        if result["status"] == "success":
            success_queries.append(result["legacy_name"])
            # Format to match your downstream logic from Cell 13
            result_list.append((result["data"], result["legacy_name"], result["rule_type_id"]))
            print(f"Success: {result['legacy_name']}")
        else:
            error_queries.append(result["rule_id"])
            print(f"Error: {result['legacy_name']}")

print(f"Total Success: {len(success_queries)}")
print(f"Total Errors: {len(error_queries)}")

The root problem is that regular expressions (Regex) are a fundamentally flawed tool for parsing SQL. Regex looks at raw text patterns, meaning a simple line break, a nested subquery, or an unexpected table alias will break the regex find-and-replace completely.
To stop throwing out queries and fix them automatically, you can implement one of two advanced patterns:
1. Semantic SQL Parsing (The Abstract Syntax Tree Method)
Instead of using Regex to blindly replace text strings, you use a Python SQL parsing library like sqlglot. This engine reads the SQL and breaks it down into an Abstract Syntax Tree (AST)—a structural map of the query recognizing what is a table, what is a column, and what is an alias.
You can write a Python function to traverse this tree, say "find every node that represents a table, and if the table name is DRG_CLAIMSUMMARY, rename it to mldev.uhcumr_gold.DRG_ClaimSummary_mart." 2. The Automated Feedback Loop (Self-Healing Code)
When spark.sql() fails, it doesn't just crash; it returns a specific AnalysisException error message (e.g., COLUMN_NOT_FOUND: 'MCC_COUNT'). You can build a system that catches that exact error string, parses it to find out why it failed, and applies a targeted fix.
You would maintain a "Data Dictionary Mapping" table that maps legacy column names to their new data lake equivalents.
The loop logic would look like this:
 * Attempt spark.sql(query).
 * If it fails, extract the missing column/table name from the error message.
 * Look up that missing column in your mapping dictionary to find the new name.
 * Inject the new name into the SQL string and dynamically retry the execution.
 * If it passes on the second try, save the updated query permanently.

healed_count = 0
